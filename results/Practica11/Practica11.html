Ejercicio 1 — Renombrar las cabeceras del FASTA de proteínas La idea es: el archivo de Uniprot descargado vive en data/, el resultado limpio (proteinas.fasta) vive en la carpeta de trabajo de esta práctica, y el chunk solo lo regenera si todavía no existe.

set +e
set +o pipefail 2>/dev/null


DATA=/mnt/c/Documents/Bioinformatica/data/Practica11
WORK=/mnt/c/Documents/Bioinformatica/results/Practica11


INPUT="$DATA/uniprotkb_proteome_UP000006548_AND_prot_2026_05_17.fasta"
OUTPUT="$WORK/proteinas.fasta"

mkdir -p "$WORK"

if [ ! -f "$OUTPUT" ]; then
    if [ ! -f "$INPUT" ]; then
        echo "ERROR: no encuentro el FASTA de Uniprot en $INPUT"
        true; exit 0
    fi
    echo "Generando $OUTPUT a partir de $INPUT ..."
    awk '(/^>/){split($1, A, /\|/); print ">" A[2]} (/^[^>]/){print $0}' \
        "$INPUT" > "$OUTPUT"
    echo "Hecho."
else
    echo "Ya existe $OUTPUT, no se vuelve a generar."
fi

# Comprobación rápida
echo "Primeras 4 líneas:"
head -n 4 "$OUTPUT"
echo "Número de proteínas: $(grep -c '^>' "$OUTPUT")"
